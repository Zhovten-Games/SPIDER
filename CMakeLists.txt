cmake_minimum_required(VERSION 3.20)
project(SPIDER LANGUAGES CXX)
add_library(spider INTERFACE)
target_include_directories(spider INTERFACE include)
target_compile_features(spider INTERFACE cxx_std_20)
include(CTest)
if(BUILD_TESTING)
  add_executable(spider-contracts tests/contracts.cpp)
  target_link_libraries(spider-contracts PRIVATE spider)
  add_test(NAME spider-contracts COMMAND spider-contracts)
endif()
